{%- comment -%} One publication link with an inline icon (Lucide-style strokes, so no icon font has to load). The icon is chosen from the label in _data/publications.json; unknown labels get a plain link icon. Labels such as "Video(Youtube)" or "CODE" are shown as "Video" and "Code". Usage: include publication-link.html url=link.url label=link.label {%- endcomment -%} {%- assign label = include.label | default: 'Link' -%} {%- assign key = label | downcase -%} {%- assign text = label -%} {%- if key contains 'video' or key contains 'youtube' -%} {%- assign kind = 'video' -%}{%- assign text = 'Video' -%} {%- elsif key contains 'code' or key contains 'github' -%} {%- assign kind = 'code' -%}{%- assign text = 'Code' -%} {%- elsif key contains 'project' or key contains 'demo' or key contains 'website' -%} {%- assign kind = 'project' -%} {%- elsif key contains 'paper' or key contains 'pdf' or key contains 'arxiv' or key contains 'doi' -%} {%- assign kind = 'paper' -%} {%- elsif key contains 'survey' or key contains 'question' or key contains 'data' or key contains 'supplement' -%} {%- assign kind = 'list' -%} {%- else -%} {%- assign kind = 'link' -%} {%- endif -%} {%- assign first_character = include.url | slice: 0 -%} {{ text }}
